    df["next_quarter_index"] - df["quarter_index"]
) == 1

    (df["running_ever_adopted"] == 0)
    & df["window_is_contiguous"]
    & df["gap_to_next_is_one"]
    & df["next_brand_prescribed"].notna()
)

# 11. TRAIN-ONLY CORRELATION FILTER
#

raw_numeric_train = X_train_raw.select_dtypes(
    include=["number"]
).copy()

raw_numeric_train = raw_numeric_train.replace(
    [np.inf, -np.inf],
    np.nan
)

raw_numeric_train = raw_numeric_train.fillna(
    raw_numeric_train.median()
)

# Remove constants.
raw_numeric_train = raw_numeric_train.loc[
    :,
    raw_numeric_train.nunique() > 1
]

target_corr_raw = {}

for col in raw_numeric_train.columns:
    corr_value = raw_numeric_train[col].corr(y_train)

    target_corr_raw[col] = (
        0 if pd.isna(corr_value)
        else abs(corr_value)
    )

corr_matrix = raw_numeric_train.corr().abs()

upper = corr_matrix.where(
    np.triu(
        np.ones(corr_matrix.shape),
        k=1
    ).astype(bool)
)

drop_corr = set()

for col in upper.columns:
    highly_correlated = upper.index[
        upper[col] >= 0.85
    ].tolist()

    for other in highly_correlated:
        if target_corr_raw[col] >= target_corr_raw[other]:
            drop_corr.add(other)
        else:
            drop_corr.add(col)

corr_kept_raw = [
    c for c in raw_numeric_train.columns
    if c not in drop_corr
]

print("Raw numeric features before correlation filter:",
      len(raw_numeric_train.columns))
print("Dropped because |correlation| >= 0.85:",
      len(drop_corr))
print("Raw numeric features after correlation filter:",
      len(corr_kept_raw))

# 12. TRAIN-ONLY VIF
# ============================================================

print("\n" + "=" * 90)
print("TRAIN-ONLY VIF")
print("=" * 90)

vif_data = raw_numeric_train[corr_kept_raw].copy()

# Iterative VIF removal is more robust than one-pass removal.
vif_threshold = 10.0
vif_removed = []

while len(vif_data.columns) > 1:
    vif_records = []

    for i, col in enumerate(vif_data.columns):
        try:
            vif_value = variance_inflation_factor(
                vif_data.values,
                i
            )
        except Exception:
            vif_value = np.inf

        vif_records.append({
            "feature": col,
            "VIF": vif_value
        })

    vif_current = (
        pd.DataFrame(vif_records)
        .sort_values("VIF", ascending=False)
    )

    max_vif = vif_current.iloc[0]["VIF"]
    max_feature = vif_current.iloc[0]["feature"]

    if max_vif <= vif_threshold:
        break

    vif_removed.append({
        "feature": max_feature,
        "VIF": max_vif
    })

    vif_data = vif_data.drop(
        columns=[max_feature]
    )

vif_df = (
    pd.DataFrame(vif_removed)
    if vif_removed
    else pd.DataFrame(
        columns=["feature", "VIF"]
    )
)

print("\nVIF removals:")
print(vif_df.round(4))

raw_numeric_survivors = vif_data.columns.tolist()

print(
    "\nNumeric features surviving correlation + VIF:",
    len(raw_numeric_survivors)
)
print(raw_numeric_survivors)


# ============================================================
# 13. ADVANCED WINDOW FEATURE ENGINEERING
#
# Feature engineering is deliberately richer than the original.
#
# All features are computed at the CURRENT WINDOW END.
# Therefore, a feature at Q3 uses only Q1-Q3.
# The target is Q4.
#
# MOMENTUM:
#   first difference / velocity:
#       Q3 - Q1
#
#   second difference / momentum (acceleration):
#       (Q3 - Q2) - (Q2 - Q1)
#       = Q3 - 2*Q2 + Q1
#
# Positive momentum means activity is accelerating.
# Negative momentum means activity is decelerating.
# ============================================================

print("\n" + "=" * 90)
print("ADVANCED FEATURE ENGINEERING")
print("=" * 90)


def window_mean(colname):
    return grp[colname].transform(
        lambda x: x.rolling(
            WINDOW_SIZE,
            min_periods=WINDOW_SIZE
        ).mean()
    )


def window_std(colname):
    return grp[colname].transform(
        lambda x: x.rolling(
            WINDOW_SIZE,
            min_periods=WINDOW_SIZE
        ).std()
    )


def window_max(colname):
    return grp[colname].transform(
        lambda x: x.rolling(
            WINDOW_SIZE,
            min_periods=WINDOW_SIZE
        ).max()
    )


def window_min(colname):
    return grp[colname].transform(
        lambda x: x.rolling(
            WINDOW_SIZE,
            min_periods=WINDOW_SIZE
        ).min()
    )


def window_sum(colname):
    return grp[colname].transform(
        lambda x: x.rolling(
            WINDOW_SIZE,
            min_periods=WINDOW_SIZE
        ).sum()
    )


def window_first(colname):
    return grp[colname].shift(
        WINDOW_SIZE - 1
    )


def window_lag1(colname):
    return grp[colname].shift(1)


def window_trend(colname):
    return (
        df[colname]
        - window_first(colname)
    )


def window_momentum(colname):
    """
    Second-order change:
        (Q3 - Q2) - (Q2 - Q1)
        = Q3 - 2Q2 + Q1

    Positive -> accelerating upward.
    Negative -> decelerating / accelerating downward.
    """
    q1 = grp[colname].shift(WINDOW_SIZE - 1)
    q2 = grp[colname].shift(1)
    q3 = df[colname]

    return (
        (q3 - q2)
        - (q2 - q1)
    )


def window_cv(colname):
    mean_value = window_mean(colname)
    std_value = window_std(colname)

    return std_value / (
        mean_value.abs() + 1e-6
    )


def safe_ratio(numerator, denominator):
    return numerator / (
        denominator.abs() + 1e-6
    )

def cv_objective(
    trial,
    model_type,
    X_data,
    y_data
):
    if model_type == "xgboost":
        model = build_xgb_from_trial(
            trial
        )
    elif model_type == "lightgbm":
        model = build_lgbm_from_trial(
            trial
        )
    else:
        raise ValueError(
            "Unknown model type."
        )

    pipeline = ImbPipeline(
        steps=[
            (
                "preprocessor",
                preprocessor
            ),
            (
                "smote",
                SMOTE(
                    random_state=RANDOM_STATE,
                    k_neighbors=5
                )
            ),
            (
                "model",
                model
            )
        ]
    )

    scores = cross_val_score(
        pipeline,
        X_data,
        y_data,
        scoring="average_precision",
        cv=cv,
        n_jobs=1
    )

    return scores.mean()

def calculate_metrics(
    y_true,
    probability,
    threshold=0.5
):
    prediction = (
        probability >= threshold
    ).astype(int)

    cm = confusion_matrix(
        y_true,
        prediction,
        labels=[0, 1]
    )

    tn, fp, fn, tp = cm.ravel()

    specificity = (
        tn / (tn + fp)
        if (tn + fp) > 0
        else 0
    )

    sensitivity = (
        tp / (tp + fn)
        if (tp + fn) > 0
        else 0
    )

    return {
        "Accuracy":
            accuracy_score(
                y_true,
                prediction
            ),
        "Precision":
            precision_score(
                y_true,
                prediction,
                zero_division=0
            ),
        "Recall":
            recall_score(
                y_true,
                prediction,
                zero_division=0
            ),
        "Sensitivity":
            sensitivity,
        "Specificity":
            specificity,
        "F1":
            f1_score(
                y_true,
                prediction,
                zero_division=0
            ),
        "ROC_AUC":
            roc_auc_score(
                y_true,
                probability
            ),
        "PR_AUC":
            average_precision_score(
                y_true,
                probability
            ),
        "Brier_Score":
            brier_score_loss(
                y_true,
                probability
            )
    }

                X_valid
            )[:, 1],
        "Test":
            model.predict_proba(
                X_test_hist
            )[:, 1]
    }


results = []

for model_name, predictions in all_predictions.items():

    for dataset_name, y_true in [
        ("Train", y_train),
        ("Validation", y_valid),
        ("Test", y_test_hist)
    ]:

        metrics = calculate_metrics(
            y_true,
            predictions[dataset_name]
        )

        metrics["Model"] = model_name
        metrics["Dataset"] = dataset_name

        results.append(metrics)


results_df = pd.DataFrame(
    results
)[
    [
        "Model",
        "Dataset",
        "Accuracy",
        "Precision",
        "Recall",
        "Sensitivity",
        "Specificity",
        "F1",
        "ROC_AUC",
        "PR_AUC",
        "Brier_Score"
    ]
]

    results_df.round(4)

# 22. OVERFITTING ANALYSIS + MODEL SELECTION
# ============================================================

overfitting_rows = []

for model_name in all_predictions.keys():

    tr = calculate_metrics(
        y_train,
        all_predictions[model_name]["Train"]
    )

    va = calculate_metrics(
        y_valid,
        all_predictions[model_name]["Validation"]
    )

    te = calculate_metrics(
        y_test_hist,
        all_predictions[model_name]["Test"]
    )

    overfitting_rows.append({
        "Model":
            model_name,

        "Train_ROC_AUC":
            tr["ROC_AUC"],

        "Validation_ROC_AUC":
            va["ROC_AUC"],

        "Test_ROC_AUC":
            te["ROC_AUC"],

        "Train_Validation_ROC_Gap":
            tr["ROC_AUC"] - va["ROC_AUC"],

        "Validation_Test_ROC_Gap":
            va["ROC_AUC"] - te["ROC_AUC"],

        "Train_PR_AUC":
            tr["PR_AUC"],

        "Validation_PR_AUC":
            va["PR_AUC"],

        "Test_PR_AUC":
            te["PR_AUC"],

        "Train_Validation_PR_Gap":
            tr["PR_AUC"] - va["PR_AUC"],

        "Validation_Test_PR_Gap":
            va["PR_AUC"] - te["PR_AUC"]
    })


overfitting_df = pd.DataFrame(
    overfitting_rows
)

selection_df = overfitting_df.copy()

selection_df[
    "stability_penalty"
] = (
    selection_df[
        "Train_Validation_PR_Gap"
    ].abs()
    +
    selection_df[
        "Validation_Test_PR_Gap"
    ].abs()
)

selection_df[
    "selection_score"
] = (
    selection_df["Test_PR_AUC"]
    -
    0.5 *
    selection_df["stability_penalty"]
)

selection_df = selection_df.sort_values(
    "selection_score",
    ascending=False
)

print("\nModel selection:")
print(
    selection_df[
        [
            "Model",
            "Validation_PR_AUC",
            "Test_PR_AUC",
            "Train_Validation_PR_Gap",
            "Validation_Test_PR_Gap",
            "selection_score"
        ]
    ].round(4)
)

final_model_name = (
    selection_df.iloc[0]["Model"]
)

final_model = models[
    final_model_name
]

    labels=False,
    duplicates="drop"

    gains_df
    .groupby("decile")
    .agg(
        physicians=("y_true", "size"),
        adopters=("y_true", "sum")

    .reset_index()

    lift_table["adopters"]

    lift_table["physicians"]

    gains_df["y_true"].mean()

    lift_table["adoption_rate"]
    /

    else np.nan
