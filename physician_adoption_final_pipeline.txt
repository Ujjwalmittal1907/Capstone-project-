# ============================================================
# ML 102B - PHYSICIAN NEXT-QUARTER BRAND ADOPTION
# FINAL LEAKAGE-SAFE END-TO-END PIPELINE
# Models: XGBoost + LightGBM only
#
# BUSINESS QUESTION
# Predict whether a physician who has NOT yet adopted the brand
# will adopt it in the NEXT QUARTER.
#
# DESIGN PRINCIPLES
# 1. Split historical target period FIRST (latest observed quarter = test).
# 2. EDA is performed on TRAIN ONLY; TEST is never used for feature selection.
# 3. Feature engineering uses only information available BEFORE the target quarter.
# 4. 3-quarter business window captures level, momentum, engagement,
#    competitive pressure, clinical opportunity and payer mix.
# 5. physician_gender is excluded completely. physician_value_tier is excluded
#    because it can be outcome-derived / populated after sustained prescribing.
# 6. Raw variables are not blindly summed and kept. Business composites,
#    rates, shares, breadth, trend and concentration are created instead.
# 7. Correlation, VIF, mutual information and univariate/bivariate analysis
#    are used for diagnosis/selection on TRAIN ONLY.
# 8. Final feature set is capped at ~13-16 pre-one-hot features.
# 9. Within TRAIN, StratifiedKFold(3) + SMOTE + Optuna is used.
# 10. SMOTE is INSIDE every CV training fold only.
# 11. Preprocessing is also INSIDE every CV fold through the imblearn pipeline.
# 12. Recall is explicitly prioritized using F2-based threshold selection from
#     out-of-fold TRAIN predictions; TEST threshold is never optimized.
# 13. Test is evaluated once at the TRAIN-selected threshold.
# 14. SHAP is calculated only after final model selection.
# 15. Final Q11 physician predictions are saved to CSV.
# ============================================================

# ============================================================
# 5. WINDOW FEATURE ENGINEERING FUNCTION
#
# A window ending at t uses ONLY t-2, t-1, t information.
# Target is brand_prescribed at t+1.
# No target-quarter values are ever used as predictors.
# ============================================================

def safe_div(a, b):
    return a / (b + 1e-6)


def build_window_dataset(source_df, target_quarters):
    data = source_df.copy().sort_values(["physician_id", "quarter_index"]).reset_index(drop=True)

    # Do not use future adoption to construct features.
    grp = data.groupby("physician_id", sort=False)
    data["running_ever_adopted"] = grp["brand_prescribed"].cummax()
    data["next_quarter_index"] = grp["quarter_index"].shift(-1)
    data["next_brand_prescribed"] = grp["brand_prescribed"].shift(-1)
    data["prev2_quarter_index"] = grp["quarter_index"].shift(2)

    data["window_contiguous"] = (
        data["quarter_index"] - data["prev2_quarter_index"] == WINDOW_SIZE - 1
    )
    data["next_contiguous"] = (
        data["next_quarter_index"] - data["quarter_index"] == 1
    )

    # Eligible history must be non-adopted through the window and have known next quarter outcome.
    data["eligible"] = (
        (data["running_ever_adopted"] == 0)
        & data["window_contiguous"]
        & data["next_contiguous"]
        & data["next_brand_prescribed"].notna()
    )

    # Detect raw business variables.
    def col(name):
        return name if name in data.columns else None

    rep = col("total_representative_visits")
    samples = col("total_sample_dropped")
    cards = col("saving_cards_dropped")
    vouchers = col("vouchers_dropped")
    sem_attend = col("total_seminar_as_attendee")
    sem_speak = col("total_seminar_as_speaker")
    ind1 = col("total_prescriptions_for_indication1")
    ind2 = col("total_prescriptions_for_indication2")
    ind3 = col("total_prescriptions_for_indication3")
    competitor = col("total_competitor_prescription")
    new_rx = col("new_prescriptions")
    comm = col("total_patient_with_commercial_insurance_plan")
    medicare = col("total_patient_with_medicare_insurance_plan")
    medicaid = col("total_patient_with_medicaid_insurance_plan")
    web = col("brand_web_impressions")
    ehr = col("brand_ehr_impressions")
    enews = col("brand_enews_impressions")
    mobile = col("brand_mobile_impressions")
    organic = col("brand_organic_web_visits")
    paidsearch = col("brand_paidsearch_visits")

    promo_cols = [c for c in [rep, samples, cards, vouchers, sem_attend, sem_speak] if c]
    digital_cols = [c for c in [web, ehr, enews, mobile, organic, paidsearch] if c]
    indication_cols = [c for c in [ind1, ind2, ind3] if c]
    payer_cols = [c for c in [comm, medicare, medicaid] if c]

    # Per-quarter composites. They are intermediates, not final model variables.
    if promo_cols:
        data["promo_touches_q"] = data[promo_cols].fillna(0).sum(axis=1)
    if digital_cols:
        data["digital_engagement_q"] = data[digital_cols].fillna(0).sum(axis=1)
    if indication_cols:
        data["indication_rx_q"] = data[indication_cols].fillna(0).sum(axis=1)

    grp = data.groupby("physician_id", sort=False)

    def wmean(c):
        return grp[c].transform(lambda x: x.rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).mean())

    def wsum(c):
        return grp[c].transform(lambda x: x.rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).sum())

    def wtrend(c):
        return data[c] - grp[c].shift(WINDOW_SIZE - 1)

    # -------------------------
    # BUSINESS FEATURE ENGINEERING
    # -------------------------
    # 1. Promotional pressure: average level across 3 quarters + momentum.
    if "promo_touches_q" in data:
        data["promo_intensity_3q"] = wmean("promo_touches_q")
        data["promo_momentum_3q"] = wtrend("promo_touches_q")
        data["promo_active_quarters_3q"] = grp["promo_touches_q"].transform(
            lambda x: (x.fillna(0) > 0).rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).sum()
        )

    # 2. Digital engagement: breadth matters more than raw impressions alone.
    if "digital_engagement_q" in data:
        data["digital_intensity_3q"] = wmean("digital_engagement_q")
        data["digital_momentum_3q"] = wtrend("digital_engagement_q")
        data["digital_active_quarters_3q"] = grp["digital_engagement_q"].transform(
            lambda x: (x.fillna(0) > 0).rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).sum()
        )

    # 3. Clinical opportunity: indication prescribing level + trend.
    if "indication_rx_q" in data:
        data["clinical_demand_3q"] = wmean("indication_rx_q")
        data["clinical_demand_momentum_3q"] = wtrend("indication_rx_q")

    # 4. Competitor pressure relative to total indication opportunity.
    if competitor and "indication_rx_q" in data:
        comp_sum = wsum(competitor)
        ind_sum = wsum("indication_rx_q")
        data["competitor_share_3q"] = safe_div(comp_sum, comp_sum + ind_sum)
        data["competitor_pressure_3q"] = wmean(competitor)
        data["competitor_momentum_3q"] = wtrend(competitor)

    # 5. New-Rx conversion: new prescribing activity relative to clinical opportunity.
    if new_rx and "indication_rx_q" in data:
        new_sum = wsum(new_rx)
        ind_sum = wsum("indication_rx_q")
        data["new_rx_conversion_3q"] = safe_div(new_sum, ind_sum)

    # 6. Promotional efficiency: response generated per promotional touch.
    if new_rx and "promo_touches_q" in data:
        new_sum = wsum(new_rx)
        promo_sum = wsum("promo_touches_q")
        data["promo_response_efficiency_3q"] = safe_div(new_sum, promo_sum)

    # 7. Channel breadth: how many distinct channels have been active.
    if promo_cols:
        active = pd.DataFrame({c: (data[c].fillna(0) > 0).astype(int) for c in promo_cols})
        breadth_parts = []
        for c in active.columns:
            breadth_parts.append(
                grp[c].transform(lambda x: x.rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).max())
            )
        data["promo_channel_breadth_3q"] = sum(breadth_parts)

    if digital_cols:
        active = pd.DataFrame({c: (data[c].fillna(0) > 0).astype(int) for c in digital_cols})
        breadth_parts = []
        for c in active.columns:
            breadth_parts.append(
                grp[c].transform(lambda x: x.rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).max())
            )
        data["digital_channel_breadth_3q"] = sum(breadth_parts)

    # 8. KOL / representative engagement state at window end.
    if sem_speak:
        data["kol_speaker_current"] = (data[sem_speak].fillna(0) > 0).astype(int)
    if rep:
        data["rep_active_current"] = (data[rep].fillna(0) > 0).astype(int)

    # 9. Payer mix concentration: HHI captures payer dependence / diversification.
    if len(payer_cols) >= 2:
        payer_sums = {c: wsum(c) for c in payer_cols}
        payer_total = sum(payer_sums.values()) + 1e-6
        data["payer_mix_hhi_3q"] = sum((v / payer_total) ** 2 for v in payer_sums.values())
        if comm:
            data["commercial_share_3q"] = payer_sums[comm] / payer_total

    # 10. Indication concentration: identifies concentrated vs broad clinical demand.
    if len(indication_cols) >= 2:
        ind_sums = {c: wsum(c) for c in indication_cols}
        ind_total = sum(ind_sums.values()) + 1e-6
        data["indication_mix_hhi_3q"] = sum((v / ind_total) ** 2 for v in ind_sums.values())

    # 11. Overall engagement momentum: useful when several channels move together.
    if "promo_touches_q" in data and "digital_engagement_q" in data:
        data["_total_engagement_q"] = (
            data["promo_touches_q"].fillna(0)
            + data["digital_engagement_q"].fillna(0)
        )
        data["total_engagement_momentum_3q"] = (
            data["_total_engagement_q"]
            - grp["_total_engagement_q"].shift(WINDOW_SIZE - 1)
        )

    # Keep only eligible rows whose target is in the requested target-quarter set.
    out = data[
        data["eligible"]
        & data["next_quarter_index"].isin(target_quarters)
    ].copy()
    out["target"] = out["next_brand_prescribed"].astype(int)
    out["target_quarter_index"] = out["next_quarter_index"]

    return out

# Numeric imputation ONLY for diagnostics; model preprocessing remains inside CV pipeline.

    X_num_diag = X_num_diag.fillna(X_num_diag.median())
    X_num_diag = X_num_diag.loc[:, X_num_diag.nunique(dropna=False) > 1]

    # Correlation redundancy ranking.
    corr = X_num_diag.corr(method="spearman").abs()
    upper = corr.where(np.triu(np.ones(corr.shape), k=1).astype(bool))
    redundant_corr_features = [c for c in upper.columns if any(upper[c] >= CORR_THRESHOLD)]
    print("\nTRAIN-ONLY high-correlation numeric features flagged:", redundant_corr_features)

    # VIF is diagnostic, not a blind removal rule for tree models.

def is_redundant(candidate, selected, corr_df, threshold=CORR_THRESHOLD):
    for s in selected:
        if candidate in corr_df.index and s in corr_df.columns:
            val = corr_df.loc[candidate, s]
            if pd.notna(val) and abs(val) >= threshold:

    return False

def optuna_objective(trial, model_name):
    model = make_model(model_name, trial)
    pipe = ImbPipeline([
        ("preprocessor", preprocessor),
        ("smote", SMOTE(random_state=RANDOM_STATE, k_neighbors=3)),
        ("model", model)
    ])

    scores = []
    for tr_idx, va_idx in cv.split(X_train, y_train):
        X_tr = X_train.iloc[tr_idx]
        y_tr = y_train.iloc[tr_idx]
        X_va = X_train.iloc[va_idx]
        y_va = y_train.iloc[va_idx]

        pipe.fit(X_tr, y_tr)
        prob = pipe.predict_proba(X_va)[:, 1]
        scores.append(average_precision_score(y_va, prob))

    return float(np.mean(scores))

# 14. OUT-OF-FOLD TRAIN PREDICTIONS FOR RECALL-AWARE THRESHOLD
#
# This is the key change for your poor recall problem.
# We do NOT choose threshold using the historical TEST set.
# Threshold is optimized using TRAIN OOF probabilities only.

def build_best_pipeline(model_name):
    class FixedTrial:
        def __init__(self, params):
            self.params = params
        def suggest_int(self, name, *args, **kwargs):
            return self.params[name]
        def suggest_float(self, name, *args, **kwargs):
            return self.params[name]

    model = make_model(model_name, FixedTrial(best_params[model_name]))
    return ImbPipeline([
        ("preprocessor", preprocessor),
        ("smote", SMOTE(random_state=RANDOM_STATE, k_neighbors=3)),
        ("model", model)
    ])


def f2_score_manual(y_true, y_pred):
    p = precision_score(y_true, y_pred, zero_division=0)
    r = recall_score(y_true, y_pred, zero_division=0)
    beta2 = F2_BETA ** 2
    if (beta2 * p + r) == 0:
        return 0.0
    return (1 + beta2) * p * r / (beta2 * p + r)


def get_oof_probabilities(model_name):
    oof = np.zeros(len(X_train), dtype=float)
    model = build_best_pipeline(model_name)
    for tr_idx, va_idx in cv.split(X_train, y_train):
        model.fit(X_train.iloc[tr_idx], y_train.iloc[tr_idx])
        oof[va_idx] = model.predict_proba(X_train.iloc[va_idx])[:, 1]
    return oof


def choose_recall_aware_threshold(y_true, prob):
    rows = []
    # Do not choose an extremely low threshold simply to maximize recall.
    # Require a minimum precision floor while maximizing F2.
    for t in np.arange(0.10, 0.71, 0.01):
        pred = (prob >= t).astype(int)
        p = precision_score(y_true, pred, zero_division=0)
        r = recall_score(y_true, pred, zero_division=0)
        f1 = f1_score(y_true, pred, zero_division=0)
        f2 = f2_score_manual(y_true, pred)
        rows.append({"threshold": t, "precision": p, "recall": r, "f1": f1, "f2": f2})
    threshold_df = pd.DataFrame(rows)

    # Prefer thresholds that achieve reasonable precision while emphasizing recall.
    eligible = threshold_df[threshold_df["precision"] >= 0.25]
    if eligible.empty:
        eligible = threshold_df
    best = eligible.sort_values(["f2", "recall", "precision"], ascending=False).iloc[0]
    return float(best["threshold"]), threshold_df

def evaluate(y_true, prob, threshold, dataset_name, model_name):
    pred = (prob >= threshold).astype(int)
    cm = confusion_matrix(y_true, pred, labels=[0, 1])
    tn, fp, fn, tp = cm.ravel()
    specificity = tn / (tn + fp) if (tn + fp) else 0
    return {
        "Model": model_name,
        "Dataset": dataset_name,
        "Threshold": threshold,
        "Accuracy": accuracy_score(y_true, pred),
        "Precision": precision_score(y_true, pred, zero_division=0),
        "Recall": recall_score(y_true, pred, zero_division=0),
        "Sensitivity": tp / (tp + fn) if (tp + fn) else 0,
        "Specificity": specificity,
        "F1": f1_score(y_true, pred, zero_division=0),
        "F2": f2_score_manual(y_true, pred),
        "ROC_AUC": roc_auc_score(y_true, prob),
        "PR_AUC": average_precision_score(y_true, prob),
        "Brier_Score": brier_score_loss(y_true, prob),
        "TP": tp,
        "FP": fp,
        "TN": tn,
        "FN": fn
    }

        if len(g) < WINDOW_SIZE:
            continue
        last = g.iloc[-1].copy()
        dummy = last.copy()
        dummy["quarter_index"] = float(last["quarter_index"]) + 1
        dummy["year_quarter"] = "FUTURE"
        dummy["year"] = np.nan
        dummy["quarter"] = np.nan
        dummy["brand_prescribed"] = np.nan
        dummy_rows.append(dummy)
    if dummy_rows:
        data = pd.concat([data, pd.DataFrame(dummy_rows)], ignore_index=True)

    # Feature engineering helper expects a known next target, so create windows with all quarters
    # and then take the row immediately before the dummy future quarter by direct rolling logic.
    # To avoid duplicating the business formulas, use a local copy of the generic feature function
    # after temporarily setting dummy target to 0; target is discarded immediately.
    data["brand_prescribed"] = pd.to_numeric(data["brand_prescribed"], errors="coerce").fillna(0)
    max_q = sorted(data["quarter_index"].dropna().unique())
    generated = build_window_dataset(data, max_q)
    latest = generated.groupby("physician_id", as_index=False).tail(1).copy()
    return latest

# The future rows above should be available through the normal feature table for physicians
# that have complete non-adopted history. For production, score the latest complete window.
future_rows = build_future_feature_rows(df, test_ids)

# Filter to test physicians and remove any duplicate IDs.
future_rows = future_rows[future_rows["physician_id"].isin(test_ids)].copy()
future_rows = future_rows.sort_values(["physician_id", "quarter_index"]).groupby("physician_id", as_index=False).tail(1)

# If the dataset has physicians who already adopted, the business question says they are not
# eligible for new adoption prediction; report them rather than silently calling them adopters.
all_test_ids = pd.DataFrame({"physician_id": list(test_ids)})
scored = all_test_ids.merge(future_rows[["physician_id"] + feature_columns], on="physician_id", how="left")

usable = scored[feature_columns].notna().any(axis=1)
q11_prob = np.full(len(scored), np.nan)
if usable.any():
    q11_prob[usable] = production_model.predict_proba(scored.loc[usable, feature_columns])[:, 1]

scored["predicted_adoption_probability_Q11"] = q11_prob
scored["predicted_adoption_class_Q11"] = np.where(
    scored["predicted_adoption_probability_Q11"].notna(),
    (scored["predicted_adoption_probability_Q11"] >= final_threshold).astype(int),
    np.nan
)
scored["probability_rank"] = scored["predicted_adoption_probability_Q11"].rank(ascending=False, method="first")

# Priority is based on probability, not merely the binary threshold.
try:
    scored["targeting_priority"] = pd.qcut(
        scored["predicted_adoption_probability_Q11"],
        q=4,
        labels=["Low", "Medium", "High", "Very High"],
        duplicates="drop"
    )
except Exception:
    scored["targeting_priority"] = np.nan

scored = scored.sort_values("predicted_adoption_probability_Q11", ascending=False).reset_index(drop=True)

OUTPUT_FILE = "predictions_Q11_test_physicians.csv"
