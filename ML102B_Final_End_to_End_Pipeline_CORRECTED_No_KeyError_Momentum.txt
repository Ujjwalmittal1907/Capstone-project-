def wmean(c): return grp[c].transform( lambda x: x.rolling(WINDOW_SIZE,
min_periods=WINDOW_SIZE).mean() )

def wstd(c): return grp[c].transform( lambda x: x.rolling(WINDOW_SIZE,
min_periods=WINDOW_SIZE).std() )

def wsum(c): return grp[c].transform( lambda x: x.rolling(WINDOW_SIZE,
min_periods=WINDOW_SIZE).sum() )

def wmin(c): return grp[c].transform( lambda x: x.rolling(WINDOW_SIZE,
min_periods=WINDOW_SIZE).min() )

def wmax(c): return grp[c].transform( lambda x: x.rolling(WINDOW_SIZE,
min_periods=WINDOW_SIZE).max() )

def wfirst(c): return grp[c].shift(WINDOW_SIZE - 1)

def wlag1(c): return grp[c].shift(1)

def wvelocity(c): return df[c] - wfirst(c)

def wmomentum(c): # (Q3-Q2) - (Q2-Q1) = Q3 - 2Q2 + Q1 q1 =
grp[c].shift(WINDOW_SIZE - 1) q2 = grp[c].shift(1) q3 = df[c] return
(q3 - q2) - (q2 - q1)

def wcv(c): return wstd(c) / (wmean(c).abs() + 1e-6)

def add_temporal_features(source_col, prefix): ““” ONE place creates
mean/std/min/max/current/velocity/momentum/CV/recent_change. This
guarantees momentum exists before any momentum flag is evaluated. ““” if
source_col is None or source_col not in df.columns: return []

    df[source_col] = pd.to_numeric(df[source_col], errors="coerce")

    mapping = {
        f"{prefix}_window_mean": wmean(source_col),
        f"{prefix}_window_std": wstd(source_col),
        f"{prefix}_window_min": wmin(source_col),
        f"{prefix}_window_max": wmax(source_col),
        f"{prefix}_current": df[source_col],
        f"{prefix}_velocity": wvelocity(source_col),
        f"{prefix}_momentum": wmomentum(source_col),
        f"{prefix}_cv": wcv(source_col),
        f"{prefix}_recent_change": df[source_col] - wlag1(source_col)
    }

    for name, values in mapping.items():
        df[name] = values

    return list(mapping.keys())
