# ============================================================
# ML 102B DRUG ADOPTION CAPSTONE - v4
# FIXED SLIDING-WINDOW APPROACH (window = 3 quarters -> target = 4th)
#
# LOGIC (per physician, 10 quarters of history):
#   Window (Q1,Q2,Q3) -> target Q4
#   Window (Q2,Q3,Q4) -> target Q5
#   Window (Q3,Q4,Q5) -> target Q6
#   Window (Q4,Q5,Q6) -> target Q7
#   Window (Q5,Q6,Q7) -> target Q8
#   Window (Q6,Q7,Q8) -> target Q9
#   Window (Q7,Q8,Q9) -> target Q10
#   Window (Q8,Q9,Q10) -> target Q11  <-- final forward prediction,
#                                          scored for Test_physicians.csv
#
# A window is only used if the physician had NOT adopted the brand
# by the end of that window (tracked with one running "ever adopted
# so far" flag -- see Section 7). This keeps the business question
# exactly "will a never-yet-adopted physician adopt next quarter,"
# using ONLY the 3 quarters inside the window (nothing older, and
# never anything from the target quarter itself).
#
# WINDOW_SIZE is a single constant -- change it in one place if you
# want a 2- or 4-quarter window instead.
# ============================================================

duplicate_check = df1.groupby(["physician_id", "year_quarter"]).size().reset_index(name="row_count")

# ============================================================
# 4. YEAR-QUARTER -> CONTINUOUS QUARTER INDEX
# ============================================================

df1["year_quarter"] = df1["year_quarter"].astype(str)
df1["year"] = df1["year_quarter"].str[:4].astype(int)
df1["quarter"] = df1["year_quarter"].str[-2:].astype(int)
df1["quarter_index"] = df1["year"] * 4 + df1["quarter"]
df1 = df1.sort_values(["physician_id", "quarter_index"]).reset_index(drop=True)


# ============================================================
# 5. PHYSICIAN VALUE TIER -> EXCLUSION DECISION (leakage)
#
# physician_value_tier is populated only after SUSTAINED brand
# prescribing (observed directly in the raw panel: blank for
# several quarters even after a physician starts prescribing).
# For never-yet-adopted physicians this field is structurally
# uninformative-or-worse -- any signal in it is an artifact of
# the outcome we're predicting. EXCLUDED from modeling features.
# ============================================================

VALUE_TIER_COL = "physician_value_tier"
if VALUE_TIER_COL in df1.columns:
    print("\nphysician_value_tier missing % (raw panel):", round(df1[VALUE_TIER_COL].isna().mean() * 100, 2))
    print(">> Excluded from modeling features (leakage risk).")


# ============================================================
# 6. MERGE STATIC PHYSICIAN PROFILE
# ============================================================

before_merge = len(df1)
df = df1.merge(df2, on="physician_id", how="left", validate="many_to_one", suffixes=("", "_profile"))
assert before_merge == len(df), "Row count changed after physician merge."
print("\nMerged dataset shape:", df.shape)

test_ids = set(test_physicians["physician_id"])
test_adoption_check = df[df["physician_id"].isin(test_ids)].groupby("physician_id")["brand_prescribed"].max()
print("Test physicians WITH historical adoption (should be ~0):", (test_adoption_check > 0).sum())
print("Test physicians confirmed never-adopted:", (test_adoption_check == 0).sum())


# ============================================================
# 7. WINDOW ELIGIBILITY + TARGET
#
# running_ever_adopted = 1 from the first quarter a physician
# ever prescribes, onward (a running max, so it can only turn on
# once and never turns back off).
#
# A row (= the LAST quarter of a candidate 3-quarter window) is
# eligible if:
#   - running_ever_adopted == 0 at this row (physician has NEVER
#     adopted, through and including this window's last quarter
#     -- because running_ever_adopted is monotonic, this alone
#     guarantees brand_prescribed == 0 for this row AND both
#     earlier rows in the window)
#   - the two quarters before this one, plus this one, are three
#     CONSECUTIVE quarters (no gaps in the physician's panel)
#   - a next quarter exists immediately after this one (gap == 1)
#     with a known brand_prescribed value -> that becomes target
# ============================================================

df = df.sort_values(["physician_id", "quarter_index"]).reset_index(drop=True)
grp = df.groupby("physician_id")

df["running_ever_adopted"] = grp["brand_prescribed"].cummax()

df["quarter_index_2back"] = grp["quarter_index"].shift(2)
df["window_is_contiguous"] = (df["quarter_index"] - df["quarter_index_2back"]) == (WINDOW_SIZE - 1)

df["next_quarter_index"] = grp["quarter_index"].shift(-1)
df["next_brand_prescribed"] = grp["brand_prescribed"].shift(-1)
df["gap_to_next_is_one"] = (df["next_quarter_index"] - df["quarter_index"]) == 1

df["eligible_for_model"] = (
    (df["running_ever_adopted"] == 0)
    & df["window_is_contiguous"]
    & df["gap_to_next_is_one"]
    & df["next_brand_prescribed"].notna()
)

df["target"] = df["next_brand_prescribed"].astype("Int64")

print(f"\nEligible (window, target) rows before feature engineering: {df['eligible_for_model'].sum()}")


# ============================================================
# 8. WINDOW-LEVEL FEATURE ENGINEERING (curated, non-redundant)
#
# Every rolling calculation below uses window=WINDOW_SIZE with
# min_periods=WINDOW_SIZE, evaluated AT the window's last quarter
# (no artificial shifting needed -- the window intentionally DOES
# include its own last quarter, because that data is fully known
# before the target quarter begins). Only the LAST row of each
# eligible window is ever used as a feature row, and the target
# always comes from the quarter immediately AFTER the window, so
# the window's own contents never overlap with the target.
#
# Design rule: because every window is a FIXED length, sum and
# mean carry identical information (sum = WINDOW_SIZE * mean) --
# so only mean (level) and trend (last-in-window minus first-in-
# window) are kept, never both mean and sum.
# ============================================================

def safe_col(name):
    return name if name in df.columns else None

rep = safe_col("total_representative_visits")
samples = safe_col("total_sample_dropped")
cards = safe_col("saving_cards_dropped")
vouchers = safe_col("vouchers_dropped")
sem_attend = safe_col("total_seminar_as_attendee")
sem_speak = safe_col("total_seminar_as_speaker")
ind1 = safe_col("total_prescriptions_for_indication1")
ind2 = safe_col("total_prescriptions_for_indication2")
ind3 = safe_col("total_prescriptions_for_indication3")
competitor = safe_col("total_competitor_prescription")
new_rx = safe_col("new_prescriptions")
comm_ins = safe_col("total_patient_with_commercial_insurance_plan")
medicare_ins = safe_col("total_patient_with_medicare_insurance_plan")
medicaid_ins = safe_col("total_patient_with_medicaid_insurance_plan")
web = safe_col("brand_web_impressions")
ehr = safe_col("brand_ehr_impressions")
enews = safe_col("brand_enews_impressions")
mobile = safe_col("brand_mobile_impressions")
organic = safe_col("brand_organic_web_visits")
paidsearch = safe_col("brand_paidsearch_visits")

promo_cols = [c for c in [rep, samples, cards, vouchers, sem_attend, sem_speak] if c]
digital_cols = [c for c in [web, ehr, enews, mobile, organic, paidsearch] if c]
indication_cols = [c for c in [ind1, ind2, ind3] if c]
insurance_cols = [c for c in [comm_ins, medicare_ins, medicaid_ins] if c]

# --- Per-quarter composite totals first (small, meaningful set) ---
if promo_cols:
    df["promo_touches_q"] = df[promo_cols].fillna(0).sum(axis=1)
if digital_cols:
    df["digital_engagement_q"] = df[digital_cols].fillna(0).sum(axis=1)
if indication_cols:
    df["indication_prescriptions_q"] = df[indication_cols].fillna(0).sum(axis=1)


def window_mean(colname):
    """Average per quarter across the WINDOW_SIZE quarters ending at this row."""
    return grp[colname].transform(lambda x: x.rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).mean())


def window_trend(colname):
    """Last-in-window minus first-in-window: simple within-window direction."""
    first_in_window = grp[colname].shift(WINDOW_SIZE - 1)
    return df[colname] - first_in_window


def window_sum(colname):
    """Used only internally to build ratios -- not exposed as its own feature."""
    return grp[colname].transform(lambda x: x.rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).sum())


# --- A. Promotional engagement: level + direction ---------------
if "promo_touches_q" in df.columns:
    df["promo_touches_window_mean"] = window_mean("promo_touches_q")
    df["promo_touches_window_trend"] = window_trend("promo_touches_q")

# --- B. Digital engagement: level + direction --------------------
if "digital_engagement_q" in df.columns:
    df["digital_engagement_window_mean"] = window_mean("digital_engagement_q")
    df["digital_engagement_window_trend"] = window_trend("digital_engagement_q")

# --- C. Clinical demand (indication rx): level + direction -------
if "indication_prescriptions_q" in df.columns:
    df["indication_rx_window_mean"] = window_mean("indication_prescriptions_q")
    df["indication_rx_window_trend"] = window_trend("indication_prescriptions_q")

    ind_window_sum = window_sum("indication_prescriptions_q")

    # Indication-mix concentration within the window (Herfindahl):
    # is relevant prescribing spread across indications or
    # concentrated in one, over these 3 quarters?
    if indication_cols:
        denom = ind_window_sum + 1e-6
        df["indication_concentration_hhi_window"] = sum(
            (window_sum(c) / denom) ** 2 for c in indication_cols
        )

# --- D. Competitor pressure: level + share of wallet --------------
if competitor:
    df["competitor_rx_window_mean"] = window_mean(competitor)
    df["competitor_rx_window_trend"] = window_trend(competitor)

    if "indication_prescriptions_q" in df.columns:
        comp_window_sum = window_sum(competitor)
        df["competitor_share_of_wallet_window"] = (
            comp_window_sum / (comp_window_sum + ind_window_sum + 1)
        )

# --- E. New-prescription rate within the window --------------------
if new_rx and "indication_prescriptions_q" in df.columns:
    new_rx_window_sum = window_sum(new_rx)
    df["new_rx_rate_window"] = new_rx_window_sum / (ind_window_sum + 1)

# --- F. Promotional efficiency within the window ---------------------
if new_rx and "promo_touches_q" in df.columns:
    promo_window_sum = window_sum("promo_touches_q")
    df["promo_efficiency_window"] = new_rx_window_sum / (promo_window_sum + 1)

# --- G. Channel breadth across the window (diversity, not volume) ----
# Count of distinct channels with ANY activity in ANY of the 3
# window quarters -- a physician touched by many different
# channels over the window is commercially different from one
# who got one large but narrow push.
if promo_cols:
    promo_any = (df[promo_cols].fillna(0) > 0).astype(int)
    df["promo_channel_breadth_window"] = sum(
        grp[c].transform(lambda x: (x.fillna(0) > 0).rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).max())
        for c in promo_cols
    )
if digital_cols:
    df["digital_channel_breadth_window"] = sum(
        grp[c].transform(lambda x: (x.fillna(0) > 0).rolling(WINDOW_SIZE, min_periods=WINDOW_SIZE).max())
        for c in digital_cols
    )

# --- H. KOL proxy + rep-active flags: current (window-end) state ------
# Single-quarter flags describing the physician's status AT the
# window's last quarter -- legitimate "current state," not an
# aggregate, so no redundancy with the window_mean/trend features.
if sem_speak:
    df["is_kol_speaker_current"] = (df[sem_speak].fillna(0) > 0).astype(int)
if rep:
    df["rep_active_current"] = (df[rep].fillna(0) > 0).astype(int)

# --- I. Payer-mix concentration within the window -----------------------
if len(insurance_cols) >= 2:
    ins_window_sums = {c: window_sum(c) for c in insurance_cols}
    ins_total_window = sum(ins_window_sums.values()) + 1e-6
    df["payer_mix_hhi_window"] = sum((s / ins_total_window) ** 2 for s in ins_window_sums.values())
    if comm_ins:
        df["commercial_insurance_share_window"] = ins_window_sums[comm_ins] / ins_total_window

print("\nWindow feature engineering complete. Total columns in df:", df.shape[1])


# ============================================================
# 9. BUILD MODELING DATASET (only eligible window-end rows)
# ============================================================

model_data = df[df["eligible_for_model"]].copy()
model_data["target"] = model_data["next_brand_prescribed"].astype(int)

print("\nFinal modeling population (one row per eligible window):", model_data.shape)
print("Target distribution:\n", model_data["target"].value_counts())
print("Target rate: {:.2f}%".format(model_data["target"].mean() * 100))


# ============================================================
# 10. CANDIDATE FEATURE POOL (window features + static profile)
# ============================================================

leakage_cols = [
    "brand_prescribed", "next_brand_prescribed", "next_quarter_index",
    "running_ever_adopted", "quarter_index_2back", "window_is_contiguous",
    "gap_to_next_is_one", "eligible_for_model", "target"
]
identifier_and_calendar_cols = ["physician_id", "year_quarter", "year", "quarter", "quarter_index"]

# Raw per-quarter columns and the intermediate composite totals are
# dropped -- their information is already captured by the window
# mean/trend/ratio features above. Keeping both would reintroduce
# the redundancy we're specifically removing.
raw_and_intermediate_cols_to_drop = [c for c in [
    rep, samples, cards, vouchers, sem_attend, sem_speak,
    ind1, ind2, ind3, competitor, new_rx,
    comm_ins, medicare_ins, medicaid_ins,
    web, ehr, enews, mobile, organic, paidsearch,
    "promo_touches_q", "digital_engagement_q", "indication_prescriptions_q"
] if c]

excluded_cols = [VALUE_TIER_COL, "physician_gender"] + raw_and_intermediate_cols_to_drop

candidate_feature_columns = [
    c for c in model_data.columns
    if c not in leakage_cols
    and c not in identifier_and_calendar_cols
    and c not in excluded_cols
]

print("\nCandidate feature pool size (curated, pre-selection):", len(candidate_feature_columns))
print(candidate_feature_columns)


# ============================================================
# 11. TEMPORAL TRAIN / VALIDATION / TEST SPLIT
#
# Split by which quarter each window's TARGET falls in (i.e. by
# next_quarter_index) -- train on windows whose target quarter is
# earliest, validate/test on windows whose target quarter is most
# recent. Same "train on the past, test on the most recent period"
# discipline as before, just aligned to the window scheme.
# ============================================================

model_data["target_quarter_index"] = model_data["next_quarter_index"]
target_quarters = sorted(model_data["target_quarter_index"].unique())
print("\nDistinct target quarters produced by the window scheme:", target_quarters)

test_quarter = target_quarters[-1]
validation_quarter = target_quarters[-2]

train_mask = model_data["target_quarter_index"] < validation_quarter
validation_mask = model_data["target_quarter_index"] == validation_quarter
test_mask = model_data["target_quarter_index"] == test_quarter

X_train_full = model_data.loc[train_mask, candidate_feature_columns].copy()
y_train = model_data.loc[train_mask, "target"].copy()
X_valid_full = model_data.loc[validation_mask, candidate_feature_columns].copy()
y_valid = model_data.loc[validation_mask, "target"].copy()
X_test_hist_full = model_data.loc[test_mask, candidate_feature_columns].copy()
y_test_hist = model_data.loc[test_mask, "target"].copy()

print("\nTRAIN:", X_train_full.shape, "| VALIDATION:", X_valid_full.shape, "| TEST:", X_test_hist_full.shape)
print("Train/Valid/Test target rate:", round(y_train.mean(), 4), round(y_valid.mean(), 4), round(y_test_hist.mean(), 4))


# ============================================================
# 12. CORRELATION + VIF FEATURE SELECTION (fit on TRAIN only)
# ============================================================

categorical_all = X_train_full.select_dtypes(include=["object", "category", "bool"]).columns.tolist()
numerical_all = X_train_full.select_dtypes(include=["number"]).columns.tolist()

print("\nCategorical candidates:", categorical_all)

train_numeric = X_train_full[numerical_all].replace([np.inf, -np.inf], np.nan)
train_numeric = train_numeric.fillna(train_numeric.median())
train_numeric = train_numeric.loc[:, train_numeric.nunique() > 1]

target_corr = {}
for col in train_numeric.columns:
    corr_value = train_numeric[col].corr(y_train)
    target_corr[col] = 0 if pd.isna(corr_value) else abs(corr_value)

corr_matrix = train_numeric.corr().abs()
upper = corr_matrix.where(np.triu(np.ones(corr_matrix.shape), k=1).astype(bool))

drop_corr = set()
for col in upper.columns:
    for other in upper.index[upper[col] >= 0.85]:
        if target_corr[col] >= target_corr[other]:
            drop_corr.add(other)
        else:
            drop_corr.add(col)

corr_features = [c for c in train_numeric.columns if c not in drop_corr]

vif_records = []
vif_data = train_numeric[corr_features].copy()
for i, col in enumerate(vif_data.columns):
    try:
        vif_value = variance_inflation_factor(vif_data.values, i)
    except Exception:
        vif_value = np.inf
    vif_records.append({"feature": col, "VIF": vif_value})

vif_df = pd.DataFrame(vif_records).sort_values("VIF", ascending=False)
print("\nVIF diagnostics:")
print(vif_df)

high_vif = set(vif_df.loc[vif_df["VIF"] > 10, "feature"])
corr_features = [c for c in corr_features if c not in high_vif]

ranked_numeric = sorted(corr_features, key=lambda c: target_corr.get(c, 0), reverse=True)
selected_numeric = ranked_numeric[:14]
selected_categorical = categorical_all[:3]

feature_columns = selected_numeric + selected_categorical

print("\n=== FINAL feature_columns ===")
for i, col in enumerate(feature_columns, 1):
    print(i, col)

assert "physician_gender" not in feature_columns
assert VALUE_TIER_COL not in feature_columns


# ============================================================
# 13. SLICE ALL SPLITS TO FINAL FEATURE LIST
# ============================================================

X_train = X_train_full[feature_columns].copy()
X_valid = X_valid_full[feature_columns].copy()
X_test_hist = X_test_hist_full[feature_columns].copy()

numerical_features = [c for c in feature_columns if c in selected_numeric]
categorical_features = [c for c in feature_columns if c in selected_categorical]

print("\nNumerical features:", len(numerical_features), "| Categorical features:", len(categorical_features))


# ============================================================
# 14. PREPROCESSOR
# ============================================================

numeric_pipeline = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler())
])
categorical_pipeline = Pipeline(steps=[
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("onehot", OneHotEncoder(handle_unknown="ignore", sparse_output=False))
])
preprocessor = ColumnTransformer(transformers=[
    ("num", numeric_pipeline, numerical_features),
    ("cat", categorical_pipeline, categorical_features)
], remainder="drop")


# ============================================================
# 15. STRATIFIED K-FOLD + OPTUNA (XGBoost & LightGBM only)
# ============================================================

cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)


def cv_objective(trial, model_type, X_data, y_data):
    if model_type == "xgboost":
        model = XGBClassifier(
            n_estimators=trial.suggest_int("n_estimators", 100, 400, step=50),
            max_depth=trial.suggest_int("max_depth", 2, 6),
            learning_rate=trial.suggest_float("learning_rate", 0.02, 0.2, log=True),
            min_child_weight=trial.suggest_int("min_child_weight", 1, 10),
            subsample=trial.suggest_float("subsample", 0.7, 1.0),
            colsample_bytree=trial.suggest_float("colsample_bytree", 0.7, 1.0),
            gamma=trial.suggest_float("gamma", 0, 3),
            reg_alpha=trial.suggest_float("reg_alpha", 1e-4, 5, log=True),
            reg_lambda=trial.suggest_float("reg_lambda", 1e-3, 10, log=True),
            eval_metric="logloss", random_state=RANDOM_STATE, n_jobs=-1
        )
    elif model_type == "lightgbm":
        model = LGBMClassifier(
            n_estimators=trial.suggest_int("n_estimators", 100, 400, step=50),
            num_leaves=trial.suggest_int("num_leaves", 7, 40),
            max_depth=trial.suggest_int("max_depth", 2, 8),
            learning_rate=trial.suggest_float("learning_rate", 0.02, 0.2, log=True),
            min_child_samples=trial.suggest_int("min_child_samples", 10, 50),
            subsample=trial.suggest_float("subsample", 0.7, 1.0),
            colsample_bytree=trial.suggest_float("colsample_bytree", 0.7, 1.0),
            reg_alpha=trial.suggest_float("reg_alpha", 1e-4, 5, log=True),
            reg_lambda=trial.suggest_float("reg_lambda", 1e-3, 10, log=True),
            random_state=RANDOM_STATE, n_jobs=-1, verbosity=-1
        )
    else:
        raise ValueError("Unknown model type.")

    pipeline = ImbPipeline(steps=[
        ("preprocessor", preprocessor),
        ("smote", SMOTE(random_state=RANDOM_STATE, k_neighbors=5)),
        ("model", model)
    ])
    scores = cross_val_score(pipeline, X_data, y_data, scoring="average_precision", cv=cv, n_jobs=1)
    return scores.mean()


model_types = ["xgboost", "lightgbm"]
best_params, best_scores = {}, {}

for model_type in model_types:
    print("\n" + "=" * 70, "\nOPTUNA:", model_type, "\n" + "=" * 70)
    study = optuna.create_study(direction="maximize", sampler=optuna.samplers.TPESampler(seed=RANDOM_STATE))
    study.optimize(lambda trial: cv_objective(trial, model_type, X_train, y_train), n_trials=15, show_progress_bar=True)
    best_params[model_type] = study.best_params
    best_scores[model_type] = study.best_value
    print("Best CV PR-AUC:", study.best_value)

print("\nOptuna summary:")
print(pd.DataFrame({"model": list(best_scores), "CV_PR_AUC": list(best_scores.values())})
      .sort_values("CV_PR_AUC", ascending=False))

# 16. BUILD + FIT FINAL TUNED PIPELINES
# ============================================================

def create_model(model_type, params):
    if model_type == "xgboost":
        return XGBClassifier(**params, eval_metric="logloss", random_state=RANDOM_STATE, n_jobs=-1)
    if model_type == "lightgbm":
        return LGBMClassifier(**params, random_state=RANDOM_STATE, n_jobs=-1, verbosity=-1)
    raise ValueError("Unknown model type.")


models = {}
for model_type in model_types:
    model = create_model(model_type, best_params[model_type].copy())
    models[model_type] = ImbPipeline(steps=[
        ("preprocessor", preprocessor),
        ("smote", SMOTE(random_state=RANDOM_STATE, k_neighbors=5)),
        ("model", model)
    ])

for name, model in models.items():
    print("\nTraining:", name)
    model.fit(X_train, y_train)

print("\nAll models trained successfully.")


# ============================================================
# 17. HISTORICAL BACKTEST EVALUATION
#     (performance on the most recent PAST window-target quarter
#     -- NOT the Q11 forward prediction. See Section 21 for that.)
# ============================================================

def calculate_metrics(y_true, probability, threshold=0.5):
    prediction = (probability >= threshold).astype(int)
    cm = confusion_matrix(y_true, prediction, labels=[0, 1])
    tn, fp, fn, tp = cm.ravel()
    specificity = tn / (tn + fp) if (tn + fp) > 0 else 0
    sensitivity = tp / (tp + fn) if (tp + fn) > 0 else 0
    return {
        "Accuracy": accuracy_score(y_true, prediction),
        "Precision": precision_score(y_true, prediction, zero_division=0),
        "Recall": recall_score(y_true, prediction, zero_division=0),
        "Sensitivity": sensitivity, "Specificity": specificity,
        "F1": f1_score(y_true, prediction, zero_division=0),
        "ROC_AUC": roc_auc_score(y_true, probability),
        "PR_AUC": average_precision_score(y_true, probability),
        "Brier_Score": brier_score_loss(y_true, probability)
    }


all_predictions = {}
for name, model in models.items():
    all_predictions[name] = {
        "Train": model.predict_proba(X_train)[:, 1],
        "Validation": model.predict_proba(X_valid)[:, 1],
        "Test": model.predict_proba(X_test_hist)[:, 1]
    }

results = []
for model_name, predictions in all_predictions.items():
    for dataset_name, y_true in [("Train", y_train), ("Validation", y_valid), ("Test", y_test_hist)]:
        m = calculate_metrics(y_true, predictions[dataset_name])
        m["Model"], m["Dataset"] = model_name, dataset_name
        results.append(m)

results_df = pd.DataFrame(results)[
    ["Model", "Dataset", "Accuracy", "Precision", "Recall", "Sensitivity",
     "Specificity", "F1", "ROC_AUC", "PR_AUC", "Brier_Score"]
]
print("\nModel performance @ threshold 0.50 (historical backtest):")
print(results_df.round(4))

auc_comparison = results_df.pivot(index="Model", columns="Dataset", values="ROC_AUC")
print("\nROC-AUC comparison:\n", auc_comparison.round(4))
prauc_comparison = results_df.pivot(index="Model", columns="Dataset", values="PR_AUC")
print("\nPR-AUC comparison:\n", prauc_comparison.round(4))


# ============================================================
# 18. OVERFITTING ANALYSIS + MODEL SELECTION
# ============================================================

overfitting_rows = []
for model_name in all_predictions.keys():
    tr = calculate_metrics(y_train, all_predictions[model_name]["Train"])
    va = calculate_metrics(y_valid, all_predictions[model_name]["Validation"])
    te = calculate_metrics(y_test_hist, all_predictions[model_name]["Test"])
    overfitting_rows.append({
        "Model": model_name,
        "Train_ROC_AUC": tr["ROC_AUC"], "Validation_ROC_AUC": va["ROC_AUC"], "Test_ROC_AUC": te["ROC_AUC"],
        "Train_Validation_ROC_Gap": tr["ROC_AUC"] - va["ROC_AUC"],
        "Validation_Test_ROC_Gap": va["ROC_AUC"] - te["ROC_AUC"],
        "Train_PR_AUC": tr["PR_AUC"], "Validation_PR_AUC": va["PR_AUC"], "Test_PR_AUC": te["PR_AUC"],
        "Train_Validation_PR_Gap": tr["PR_AUC"] - va["PR_AUC"],
        "Validation_Test_PR_Gap": va["PR_AUC"] - te["PR_AUC"],
    })
overfitting_df = pd.DataFrame(overfitting_rows)

selection_df = overfitting_df.copy()
selection_df["stability_penalty"] = (
    selection_df["Train_Validation_PR_Gap"].abs() + selection_df["Validation_Test_PR_Gap"].abs()
)
selection_df["selection_score"] = selection_df["Test_PR_AUC"] - 0.5 * selection_df["stability_penalty"]
selection_df = selection_df.sort_values("selection_score", ascending=False)

print("\nModel selection:")
print(selection_df[["Model", "Validation_PR_AUC", "Test_PR_AUC",
                     "Train_Validation_PR_Gap", "Validation_Test_PR_Gap", "selection_score"]].round(4))

final_model_name = selection_df.iloc[0]["Model"]
final_model = models[final_model_name]
print("\nSelected final model:", final_model_name)

thresholds = np.arange(0.10, 0.91, 0.01)
threshold_metrics = [{
    "threshold": t,
    "precision": precision_score(y_test_hist, (final_test_prob >= t).astype(int), zero_division=0),
    "recall": recall_score(y_test_hist, (final_test_prob >= t).astype(int), zero_division=0),
    "f1": f1_score(y_test_hist, (final_test_prob >= t).astype(int), zero_division=0)
} for t in thresholds]
threshold_df = pd.DataFrame(threshold_metrics)
best_threshold = threshold_df.sort_values("f1", ascending=False).iloc[0]["threshold"]
print("\nSelected best_threshold (max F1 on historical Test):", round(best_threshold, 4))

gains_df = pd.DataFrame({"y_true": y_test_hist.values, "prob": final_test_prob}).sort_values(
    "prob", ascending=False).reset_index(drop=True)

lift_table = gains_df.groupby("decile").agg(physicians=("y_true", "size"), adopters=("y_true", "sum")).reset_index()
lift_table["adoption_rate"] = lift_table["adopters"] / lift_table["physicians"]
overall_rate = gains_df["y_true"].mean()
lift_table["lift"] = lift_table["adoption_rate"] / overall_rate if overall_rate > 0 else np.nan
print("\nLift by probability decile (0 = highest predicted probability):")

print(f"Top decile captures {lift_table.iloc[0]['adopters']:.0f} of {gains_df['y_true'].sum():.0f} adopters.")
